% Screenshot with caption; skipped if the image file isn't there.
\newcommand{\shot}[2]{%
  \IfFileExists{\imgdir/#1.jpg}{%
    \begin{figure}[htbp]\centering
      \fcolorbox{gray!40}{white}{\includegraphics[width=0.96\textwidth]{\imgdir/#1.jpg}}
      \caption{#2}
    \end{figure}}{}}

% Two-column tables that fill the text width.
\newenvironment{keytable}[2]{%
  \renewcommand{\arraystretch}{1.3}%
  \rowcolors{2}{white}{violetlight}%
  \tabularx{\textwidth}{@{}>{\raggedright\arraybackslash}p{0.33\textwidth}X@{}}
  \toprule \textsf{\textbf{#1}} & \textsf{\textbf{#2}} \\ \midrule}{%
  \bottomrule\endtabularx}
